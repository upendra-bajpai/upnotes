    const generateBtn = document.getElementById('generate-review-btn');

    const codeInput = document.getElementById('code-input');

    const reviewOutput = document.getElementById('review-output');

    const loadingSpinner = document.getElementById('loading-spinner');

    const placeholderText = document.getElementById('placeholder-text');


    generateBtn.addEventListener('click', async () => {

        const userCode = codeInput.value;

        if (!userCode.trim()) {

            reviewOutput.textContent = 'Please enter some code to review.';

            return;

        }


        placeholderText.classList.add('hidden');

        reviewOutput.textContent = '';

        loadingSpinner.classList.remove('hidden');

        generateBtn.disabled = true;


        const prompt = `You are an expert AI code review assistant. Your goal is to provide a concise, constructive, and friendly code review for the following Python snippet. Focus on identifying potential bugs, style issues (following PEP 8), performance improvements, and lack of clarity. Structure your feedback with clear headings for each point you raise.


Here is the code:

\`\`\`python

${userCode}

\`\`\``;


        let chatHistory = [{ role: "user", parts: [{ text: prompt }] }];

        const payload = { contents: chatHistory };

        const apiKey = ""; 

        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${apiKey}`;


        try {

            const response = await fetch(apiUrl, {

                method: 'POST',

                headers: { 'Content-Type': 'application/json' },

                body: JSON.stringify(payload)

            });


            if (!response.ok) {

                 throw new Error(`API Error: ${response.statusText} (Status: ${response.status})`);

            }


            const result = await response.json();

            

            let text = 'Could not retrieve a review. Please try again.';

            if (result.candidates && result.candidates.length > 0 &&

                result.candidates[0].content && result.candidates[0].content.parts &&

                result.candidates[0].content.parts.length > 0) {

              text = result.candidates[0].content.parts[0].text;

            }

            reviewOutput.textContent = text;


        } catch (error) {

            console.error("Error calling Gemini API:", error);

            reviewOutput.textContent = `An error occurred: ${error.message}. Please check the console for more details.`;

        } finally {

            loadingSpinner.classList.add('hidden');

            generateBtn.disabled = false;

        }

    });
